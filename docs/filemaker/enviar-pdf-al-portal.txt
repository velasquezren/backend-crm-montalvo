  7 Establecer variable [ Value: "<CREDENCIAL>" ; Repetition: 1 ; $tokenCRM ]
  8 Establecer variable [ Value: Ecografia::Nombre ; Repetition: 1 ; $nombre ]
  9 Establecer variable [ Value: Ecografia::rk.pac ; Repetition: 1 ; $pac ]

 14 # (comentario) [ Text: Referencia estable del estudio: con ella, volver a enviar reemplaza el PDF y no crea otro informe. No cambiarla: los informes ya enviados usan esta. ]
 15 Establecer variable [ Value: "FM-Ecografia-" & Get ( RecordID ) ; Repetition: 1 ; $referencia ]

Right ( "0" & Day ( Ecografia::Fecha ) ; 2 ) ; Repetition: 1 ; $fecha ]

 27 Establecer variable [ Value: Get ( LastError ) ; Repetition: 1 ; $errorArchivo ]
 28 If [ $errorArchivo ≠ 0 or IsEmpty ( $pdf ) ]
 29     Mostrar cuadro de diálogo personalizado [ Title: "No se pudo leer el PDF" ; Message: "Error de FileMaker: " & $errorArchivo & ". No se envió nada." ; botones: , , ]

 31 End If [  ]

 33 Establecer variable [ Value: "" ; Repetition: 1 ; $r ]

& " --form-string " & Quote ( "pac=" & $pac )
& " --form-string " & Quote ( "estudio=" & $estudio )
& " --form-string " & Quote ( "fechaEstudio=" & $fecha )
& " --form-string " & Quote ( "referencia=" & $referencia )
& " --form-string " & Quote ( "publicar=true" )
& " --form " & Quote ( "archivo=@$pdf;type=application/pdf;filename=informe.pdf" )
& " --connect-timeout 10 --max-time 120 --show-error" ; "https://resultados.107.175.132.15.nip.io/v1/integraciones/filemaker/informe" ; Text: ; destino $r ]
 35 Establecer variable [ Value: Get ( LastError ) ; Repetition: 1 ; $errorEnvio ]

and JSONGetElement ( $r ; "estado" ) = "PUBLICADO"
and not IsEmpty ( JSONGetElement ( $r ; "informeId" ) ) ; Repetition: 1 ; $publicado ]
 37 If [ not $publicado ]
 38     Establecer variable [ Value: JSONGetElement ( $r ; "error.mensaje" ) ; Repetition: 1 ; $motivo ]
 39     Mostrar cuadro de diálogo personalizado [ Title: "No se publicó el informe" ; Message: If ( IsEmpty ( $motivo ) or Left ( $motivo ; 1 ) = "?" ; "Error de FileMaker: " & $errorEnvio & "¶" & Left ( $r ; 500 ) ; $motivo ) & "¶¶No se envió nada al paciente." ; botones: , , ]
 40     Salir del guión [ $r ]

 42 # (comentario) [ Text: 2. Aviso por WhatsApp desde el CRM (solo con la credencial puesta). El CRM decide a qué número, y no manda dos veces el mismo informe. ]

 44 Establecer variable [ Value: "" ; Repetition: 1 ; $telefono ]
 45 Establecer variable [ Value: "El informe quedó publicado en el portal. Envía el aviso por WhatsApp desde Resultados en el CRM." ; Repetition: 1 ; $aviso ]
 46 If [ Length ( $tokenCRM ) ≥ 32 ]
 47     # (comentario) [ Text: 2a. Antes de enviar, el CRM dice a qué ficha y a qué número iría el aviso. No envía nada. ]
 48     Establecer variable [ Value: "" ; Repetition: 1 ; $rDestino ]
 49     Insertar desde URL [ NoInteract=True ; DontEncodeURL=False ; SelectAll=True ; VerifySSLCertificates=True ; cURL: "--request POST"
& " --header " & Quote ( "Authorization: Bearer " & $tokenCRM )
& " --connect-timeout 10 --max-time 30 --show-error" ; "https://crm.107.175.132.15.nip.io/integraciones/filemaker/resultados/" & JSONGetElement ( $r ; "informeId" ) & "/destino" ; Text: ; destino $rDestino ]
 50     Establecer variable [ Value: Get ( LastError ) ; Repetition: 1 ; $errorDestino ]
 51     Establecer variable [ Value: If ( $errorDestino = 0 ; JSONGetElement ( $rDestino ; "codigo" ) ; "" ) ; Repetition: 1 ; $codigoDestino ]
 52     If [ $codigoDestino = "LISTO" ]
 53         Establecer variable [ Value: JSONGetElement ( $rDestino ; "telefono" ) ; Repetition: 1 ; $telefonoDestino ]
 54         Mostrar cuadro de diálogo personalizado [ Title: "Enviar por WhatsApp" ; Message: "Se enviará el aviso del informe por WhatsApp a:¶¶" & JSONGetElement ( $rDestino ; "paciente" ) & "¶" & If ( IsEmpty ( $telefonoDestino ) ; "(su usuario de WhatsApp, sin número)" ; $telefonoDestino ) & "¶¶Paciente de esta ecografía: " & $nombre & "¶¶¿Es el número correcto?" ; botones: , , ]
 55         If [ Get ( LastMessageChoice ) = 1 ]
 56             # (comentario) [ Text: 2b. Envío. Va el número confirmado: si la ficha cambió entretanto, el CRM no envía. ]
 57             Establecer variable [ Value: JSONSetElement ( "{}" ;
[ "usuario" ; Get ( AccountName ) ; JSONString ] ;
[ "telefonoConfirmado" ; $telefonoDestino ; JSONString ] ) ; Repetition: 1 ; $cuerpoCRM ]
 58             Establecer variable [ Value: "" ; Repetition: 1 ; $rCRM ]
 59             Insertar desde URL [ NoInteract=True ; DontEncodeURL=False ; SelectAll=True ; VerifySSLCertificates=True ; cURL: "--request POST"
& " --header " & Quote ( "Authorization: Bearer " & $tokenCRM )
& " --header " & Quote ( "Content-Type: application/json" )
& " --data-binary @$cuerpoCRM"
& " --connect-timeout 10 --max-time 60 --show-error" ; "https://crm.107.175.132.15.nip.io/integraciones/filemaker/resultados/" & JSONGetElement ( $r ; "informeId" ) & "/avisar" ; Text: ; destino $rCRM ]
 60             Establecer variable [ Value: Get ( LastError ) ; Repetition: 1 ; $errorCRM ]
 61             Establecer variable [ Value: If ( $errorCRM = 0 ; JSONGetElement ( $rCRM ; "codigo" ) ; "" ) ; Repetition: 1 ; $codigo ]
 62             If [ $codigo = "ENVIADO" ]
 63                 Establecer variable [ Value: "ENVIADO" ; Repetition: 1 ; $estadoEnvio ]
 64                 Establecer variable [ Value: JSONGetElement ( $rCRM ; "telefono" ) ; Repetition: 1 ; $telefono ]
 65                 Establecer variable [ Value: "WhatsApp enviado al paciente" & If ( IsEmpty ( $telefono ) ; " (a su usuario de WhatsApp)." ; " al " & $telefono & "." ) ; Repetition: 1 ; $aviso ]
 66             Else If [ $codigo = "YA_ENVIADO" ]
 67                 Establecer variable [ Value: "YA ENVIADO" ; Repetition: 1 ; $estadoEnvio ]
 68                 Establecer variable [ Value: JSONGetElement ( $rCRM ; "telefono" ) ; Repetition: 1 ; $telefono ]
 69                 Establecer variable [ Value: "El paciente ya tenía el aviso de este informe; no se mandó otro WhatsApp. El PDF del portal quedó actualizado." ; Repetition: 1 ; $aviso ]

 71                 Establecer variable [ Value: "El informe quedó publicado, pero NO se envió el WhatsApp:¶" & JSONGetElement ( $rCRM ; "mensaje" ) ; Repetition: 1 ; $aviso ]
 72             Else [  ]
 73                 Establecer variable [ Value: "El informe quedó publicado, pero no se pudo pedir el WhatsApp al CRM (error " & $errorCRM & "). Envíalo desde Resultados en el CRM.¶" & Left ( $rCRM ; 300 ) ; Repetition: 1 ; $aviso ]

 75         Else [  ]
 76             Establecer variable [ Value: "No se envió el WhatsApp; el informe quedó publicado. Si el número está mal, corrígelo en Resultados del CRM («Corregir teléfono») y vuelve a pulsar el botón." ; Repetition: 1 ; $aviso ]
 77         End If [  ]
 78     Else If [ $codigoDestino = "YA_ENVIADO" ]
 79         Establecer variable [ Value: "YA ENVIADO" ; Repetition: 1 ; $estadoEnvio ]
 80         Establecer variable [ Value: JSONGetElement ( $rDestino ; "telefono" ) ; Repetition: 1 ; $telefono ]
 81         Establecer variable [ Value: "El paciente ya tenía el aviso de este informe; no se mandó otro WhatsApp. El PDF del portal quedó actualizado." ; Repetition: 1 ; $aviso ]

 83         Establecer variable [ Value: "El informe quedó publicado, pero NO se envió el WhatsApp:¶" & JSONGetElement ( $rDestino ; "mensaje" ) ; Repetition: 1 ; $aviso ]

 88 # (comentario) [ Text: 3. Campos de envío. Si otra persona tiene abierto el registro no se pueden guardar: el envío ya está hecho y solo se avisa. ]
