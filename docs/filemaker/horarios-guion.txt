# CRM - Sincronizar horario actual
# Crear estos pasos en el espacio de trabajo de guiones; no es un archivo importable.
# Pegar el cálculo horarios-datos-calculo.txt en el valor del paso $datos.
# La presentación debe estar basada en horario_medicos (puede tener otro nombre de ocurrencia).

If [ $$CRM_HORARIO_EN_CURSO or Get ( WindowMode ) ≠ 0 ]
    Salir del guion [ Resultado de texto: 1 ]
End If
Establecer captura de errores [ Activo ]
Permitir cancelación del usuario [ Inactivo ]
Establecer variable [ $$CRM_HORARIO_EN_CURSO ; Valor: 1 ]
Establecer variable [ $tabla ; Valor: Get ( LayoutTableName ) ]
Establecer variable [ $token ; Valor: "PEGAR_TOKEN_PRIVADO_DE_HORARIOS" ]
Establecer variable [ $url ; Valor: "https://crm.107.175.132.15.nip.io/integraciones/filemaker/horarios" ]
Establecer variable [ $datos ; Valor: PEGAR_AQUI_EL_CALCULO_DEL_ARCHIVO_horarios-datos-calculo.txt ]

If [ IsEmpty ( JSONGetElement ( $datos ; "registroId" ) ) or IsEmpty ( JSONGetElement ( $datos ; "medicoCodigo" ) ) ]
    Mostrar cuadro de diálogo personalizado [ "Horario sin identificar" ; "Falta el identificador del horario o el código del médico. No se envió nada." ]
    Establecer variable [ $$CRM_HORARIO_EN_CURSO ; Valor: "" ]
    Salir del guion [ Resultado de texto: 0 ]
End If

Establecer variable [ $anterior ; Valor: GetField ( $tabla & "::crm_horario_cuerpo" ) ]
Establecer variable [ $cambio ; Valor:
    JSONFormatElements ( $datos ) ≠ JSONFormatElements ( JSONDeleteElement ( $anterior ; "revision" ) )
]
If [ not $cambio and GetField ( $tabla & "::crm_horario_estado" ) = "SINCRONIZADO" ]
    Establecer variable [ $$CRM_HORARIO_EN_CURSO ; Valor: "" ]
    Salir del guion [ Resultado de texto: 1 ]
End If

Establecer variable [ $error ; Valor: 0 ]
If [ $cambio ]
    Establecer variable [ $revision ; Valor: GetAsNumber ( GetField ( $tabla & "::crm_horario_revision" ) ) + 1 ]
    Establecer variable [ $cuerpo ; Valor: JSONSetElement ( $datos ; "revision" ; $revision ; JSONNumber ) ]
    Establecer campo por nombre [ $tabla & "::crm_horario_revision" ; $revision ]
    Establecer variable [ $error ; Valor: Max ( $error ; Get ( LastError ) ) ]
    Establecer campo por nombre [ $tabla & "::crm_horario_cuerpo" ; $cuerpo ]
    Establecer variable [ $error ; Valor: Max ( $error ; Get ( LastError ) ) ]
Else
    Establecer variable [ $cuerpo ; Valor: $anterior ]
    Establecer variable [ $revision ; Valor: JSONGetElement ( $cuerpo ; "revision" ) ]
End If
Establecer campo por nombre [ $tabla & "::crm_horario_estado" ; "PENDIENTE" ]
Establecer variable [ $error ; Valor: Max ( $error ; Get ( LastError ) ) ]
Establecer campo por nombre [ $tabla & "::crm_horario_error" ; "" ]
Establecer variable [ $error ; Valor: Max ( $error ; Get ( LastError ) ) ]
If [ $error ≠ 0 ]
    Mostrar cuadro de diálogo personalizado [ "No se preparó el envío" ; "Revise los cinco campos CRM y sus permisos. Error: " & $error ]
    Establecer variable [ $$CRM_HORARIO_EN_CURSO ; Valor: "" ]
    Salir del guion [ Resultado de texto: 0 ]
End If

# El activador se ejecuta ANTES de guardar. Primero se guarda explícitamente,
# sin omitir validaciones ni forzar conflictos. La variable global evita recursión.
Consignar registros/peticiones [ Con diálogo: Inactivo ]
Establecer variable [ $error ; Valor: Get ( LastError ) ]
If [ $error ≠ 0 ]
    Mostrar cuadro de diálogo personalizado [ "No se guardó el horario" ; "No se envió al CRM. Revise la validación del registro. Error: " & $error ]
    Establecer variable [ $$CRM_HORARIO_EN_CURSO ; Valor: "" ]
    Salir del guion [ Resultado de texto: 0 ]
End If

Establecer variable [ $r ; Valor: "" ]
Establecer variable [ $errorEnvio ; Valor: 0 ]
If [ Length ( $token ) < 32 ]
    Establecer variable [ $r ; Valor: "Falta configurar la credencial de horarios." ]
Else
    Insertar desde URL [ Seleccionar ; Con diálogo: Inactivo ; Destino: $r ; $url ; Verificar certificados SSL ; Opciones de cURL:
        "--request POST"
        & " --header " & Quote ( "Authorization: Bearer " & $token )
        & " --header " & Quote ( "Content-Type: application/json" )
        & " --data-binary @$cuerpo"
        & " --connect-timeout 5 --max-time 35 --show-error"
    ]
    Establecer variable [ $errorEnvio ; Valor: Get ( LastError ) ]
End If

Establecer variable [ $confirmado ; Valor:
    $errorEnvio = 0
    and JSONGetElement ( $r ; "estado" ) = "SINCRONIZADO"
    and JSONGetElement ( $r ; "registroId" ) = JSONGetElement ( $cuerpo ; "registroId" )
    and JSONGetElement ( $r ; "medicoCodigo" ) = JSONGetElement ( $cuerpo ; "medicoCodigo" )
    and JSONGetElement ( $r ; "revision" ) = $revision
]
Establecer variable [ $error ; Valor: 0 ]
If [ $confirmado ]
    Establecer campo por nombre [ $tabla & "::crm_horario_estado" ; "SINCRONIZADO" ]
    Establecer variable [ $error ; Valor: Max ( $error ; Get ( LastError ) ) ]
    Establecer campo por nombre [ $tabla & "::crm_horario_sincronizado" ; Get ( CurrentTimestamp ) ]
    Establecer variable [ $error ; Valor: Max ( $error ; Get ( LastError ) ) ]
Else
    Establecer campo por nombre [ $tabla & "::crm_horario_error" ;
        "Error FileMaker: " & $errorEnvio & "¶" & Left ( $r ; 2000 )
    ]
    Establecer variable [ $error ; Valor: Max ( $error ; Get ( LastError ) ) ]
End If
Consignar registros/peticiones [ Con diálogo: Inactivo ]
Establecer variable [ $error ; Valor: Max ( $error ; Get ( LastError ) ) ]
Establecer variable [ $$CRM_HORARIO_EN_CURSO ; Valor: "" ]
If [ $error ≠ 0 ]
    Mostrar cuadro de diálogo personalizado [ "Revise el estado local" ; "No se pudo guardar la confirmación. El envío puede haber llegado; reintentarlo es seguro. Error: " & $error ]
    Salir del guion [ Resultado de texto: 0 ]
End If
If [ not $confirmado ]
    Mostrar cuadro de diálogo personalizado [ "Horario pendiente" ; "El horario se guardó en FileMaker, pero todavía no se confirmó en el CRM. Revise crm_horario_error y vuelva a ejecutar este guion para reintentar." ]
End If
Salir del guion [ Resultado de texto: 1 ]
